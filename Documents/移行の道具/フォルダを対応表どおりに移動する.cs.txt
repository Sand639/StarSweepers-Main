using System.Collections.Generic;
using System.IO;
using System.Linq;
using UnityEditor;
using UnityEngine;

/// <summary>
/// 【一時的に置いて使う道具。Assets に置いたままコミットしない】
/// フォルダ整理の対応表（TSV：今の場所 → 新しい場所。Assets からの相対。1行に1ファイル、タブ区切り）のとおりに、
/// Unity の移動機能（AssetDatabase.MoveAsset）でファイルを移す。参照（GUID）は変わらない。空になったフォルダは消す。
///
/// 使い方（2026/11/5 のフォルダ整理で使った。`Documents/AIの申し送り.md` の「`Assets/` のフォルダを整理した」）：
///   1. このファイルを、移動しないフォルダ（例：Assets/Scripts/UI/Editor/）に TempReorg.cs として置く
///   2. Unity.exe -batchmode -quit -projectPath … -executeMethod TempReorg.Run -reorgMap 対応表.tsv -logFile …
///   3. ログの [REORG] の行で、移動の件数と失敗の件数、ビルドに入れるシーンの場所を確かめる
///   4. TempReorg.cs と .meta を消す
/// </summary>
public static class TempReorg
{
    public static void Run()
    {
        string[] args = System.Environment.GetCommandLineArgs();
        string map = null;
        for (int i = 0; i < args.Length - 1; i++)
        {
            if (args[i] == "-reorgMap") map = args[i + 1];
        }

        List<(string from, string to)> rows = File.ReadAllLines(map)
            .Where(l => l.Contains("\t"))
            .Select(l => l.Split('\t'))
            .Select(c => ("Assets/" + c[0], "Assets/" + c[1]))
            .ToList();

        // 行き先のフォルダを先に作る
        foreach (string dir in rows.Select(r => Path.GetDirectoryName(r.to).Replace('\\', '/')).Distinct().OrderBy(d => d.Length))
        {
            EnsureFolder(dir);
        }

        int moved = 0, failed = 0;
        foreach ((string from, string to) in rows)
        {
            string error = AssetDatabase.MoveAsset(from, to);
            if (string.IsNullOrEmpty(error))
            {
                moved++;
            }
            else
            {
                failed++;
                Debug.LogError($"[REORG] 移動できなかった：{from} → {to}：{error}");
            }
        }

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();

        // 空になったフォルダを、深い所から消す
        int removed = 0;
        bool again = true;
        while (again)
        {
            again = false;
            foreach (string dir in AssetDatabase.GetSubFolders("Assets").SelectMany(AllFolders).OrderByDescending(d => d.Length).ToList())
            {
                if (!AssetDatabase.IsValidFolder(dir)) continue;
                bool empty = Directory.GetFiles(dir, "*", SearchOption.AllDirectories).All(f => f.EndsWith(".meta"));
                if (empty)
                {
                    AssetDatabase.DeleteAsset(dir);
                    Debug.Log($"[REORG] 空になったフォルダを消した：{dir}");
                    removed++;
                    again = true;
                }
            }
        }

        AssetDatabase.SaveAssets();
        AssetDatabase.Refresh();

        Debug.Log($"[REORG] 移動 {moved} 件／失敗 {failed} 件／消したフォルダ {removed} 個");
        foreach (EditorBuildSettingsScene scene in EditorBuildSettings.scenes)
        {
            Debug.Log($"[REORG] build scene: {scene.path} enabled={scene.enabled} exists={File.Exists(scene.path)}");
        }
    }

    private static IEnumerable<string> AllFolders(string root)
    {
        yield return root;
        foreach (string sub in AssetDatabase.GetSubFolders(root))
        {
            foreach (string f in AllFolders(sub)) yield return f;
        }
    }

    private static void EnsureFolder(string dir)
    {
        if (AssetDatabase.IsValidFolder(dir)) return;
        string parent = Path.GetDirectoryName(dir).Replace('\\', '/');
        EnsureFolder(parent);
        AssetDatabase.CreateFolder(parent, Path.GetFileName(dir));
    }
}
